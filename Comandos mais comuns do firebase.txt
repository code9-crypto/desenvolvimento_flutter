QuerySnapshot -> é usado para obter vários documentos
DocumentSnapshot -> é usado para obter um documento

INICIALIZANDO O BANCO
OBS.: este comando deve ser usado no main.dart dentro da função main() antes do runApp()
WidgetsFlutterBinding.ensureInitialized();
await Firebase.initializeApp();

OBTENDO OS DADOS DO BANCO
//este comando retorna um QuerySnapshot; OBS.: este resultado vem no futuro
QuerySnapshot snapshot = await FirebaseFirestore.instance.collection("coleção").get(); -> este comando está pegando tanto os documentos(ID's) quanto seus respectivos campos

LENDO OS DADOS OBTIDOS DO BANCO
snapshot.docs.forEach((d){
    print(d.data)
});

OU

for(DocumentSnapshot doc in snapshot.docs){
    print(doc.data)
}

ARMAZENANDO OS DADOS NUMA LISTA DE MAPS
List<Map> dados = [];
snapshot.docs.forEach((d){
    dados.add(d.data() as Map)
})

OU

List dados = [];
snapshot.docs.forEach((d){
    dados.add(d.data()) //assim será salvo direto como lista
})

    PESQUISANDO NA LISTA COM CRITÉRIO ESPECÍFICO
    for( int i = 0; i <= dados.length -1; i++ ){
        if(dados[i]["chave"].toString().toLowerCase().startsWith("text")){
            print(dados[i]["senderName"]);
        }    
    }

ESCREVENDO NO BANCO DE DADOS

data["chave"] = "valor";

FirebaseFirestore.instance.collection("messages").doc().set(data);
